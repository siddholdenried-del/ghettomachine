# The 303 generator's KEEP page (0.5) and the drums' BEAT generator, in the host simulator.
#   app/build/host/ghettomachine_host app/tests/gen_keep.txt OUTDIR
# gkeepT_S: track T's step S kept; gcur: KEEP's step; note2_S: T2's step S's note; genN_0: the seed
# dgenI: the drum part's generator, 0 seed 1 mutates 2 style 3 kick 4 snare 5 hats 6 perc 7 fill 8 accent; hitsN: pad N's hits (0 the bass drum), hitN_S: its step S
wait 600
turn ALGO 1
wait 30
expect trk 2
# (ALGORITHM kept the page: T2's steps; SEQ: ACID)
expect page 0
tap SEQ
wait 30
expect page 14
# a line (OCT+), then KEEP (SELECT to the fourth page): step 3 kept as it is, step 5 given a note by hand
tap OCT+
wait 60
expect gen2_0 >0
turn SELECT 3
wait 30
expect genpage 3
spin K1 2
wait 30
expect gcur 3
spin K2 1
wait 30
expect gkeep2_3 1
spin K1 2
wait 30
expect gcur 5
spin K4 -60
wait 30
expect note2_5 0
expect gkeep2_5 1
spin K4 1
wait 30
expect note2_5 36
spin K4 7
wait 30
expect note2_5 43
shot gen_keep.png
# NEW and MUTATE, three times each: step 5 is still the G, kept
tap OCT+

wait 60

wait 60

expect gkeep2_5 0
