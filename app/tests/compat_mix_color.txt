# Load a real v5 project, change one drum's source and its own level, enable
# drum Color, then save and reload into v8. The discontinued 606 kit becomes
# the 808. Its historic T2 insert is retained as opaque project data.
flash_load app/tests/fixtures/proj_v5.img
reboot
wait 600
expect projfmt 5
expect kit 1
expect insert2 2
expect insert2_0 38
expect insert2_1 88
expect insert2_2 64
expect insert2_3 110
expect pot7_0 64
expect color3_5 0

# Assign the 909 open hat to the 808 kit. K1 on its first EDIT page now
# controls the 909 Level (97). The neighboring closed hat stays with the 808.
key 20 down
wait 30
key 20 up
wait 30
tap EDIT
wait 30
turn SELECT 8
wait 30
turn K1 1
wait 30
expect model8 0
expect model7 1
expect kitcustom 1
turn SELECT -8
wait 30
expect pad 8
expect pot8_0 97
spin K1 -8
wait 30
expect pot8_0 89
expect pot7_0 64
