# presets: new
# The second factory set (0.5): six banks by kind, each sound's level trim carried by the track; a project saved with
# the first set keeps its sounds, their names and their levels.  app/build/host/ghettomachine_host app/tests/presets_new.txt OUTDIR
# presetN: track N's preset id; ptrimN: its level trim, tenths of a dB; legacyN: its sound is a first-set one from a save
wait 600

expect preset2 9
expect legacy2 0
expect legacy3 0
expect legacy4 0

wait 20

turn PRESET 1
wait 60

wait 60

wait 60
